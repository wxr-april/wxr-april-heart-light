        function generateCode() {
            var chars = '0123456789ABCDEFGHIJKLMNOPQRSTUVWXYZ';
            var code = 'HX-2026-';
            for (var i = 0; i < 4; i++) {
                code += chars.charAt(Math.floor(Math.random() * chars.length));
            }
            return code;
