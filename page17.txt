            } else if (isTokyo) {
                dotClass += ' lit special-blink';
                isSpecial = true;
            } else {
                dotClass += ' lit';
            }

            const extraClass = isSpecial ? ' special' : '';
            html += `
                <div class="light-point${extraClass}" data-name="${site.name}" data-status="${site.status}" data-is-tokyo="${isTokyo ? 'true' : 'false'}" data-is-anxi="${isAnxi ? 'true' : 'false'}">
                    <span class="${dotClass}"></span>
                    <span class="name">${site.name}</span>
                </div>
            `;
        });
        grid.innerHTML = html;
        document.getElementById('totalCount').textContent = SITES.length;
    }
    renderMap();

    // =============================================================
    // 3. 点击事件
    // =============================================================
    const overlay = document.getElementById('detailOverlay');
    const dTitle = document.getElementById('dTitle');
    const dStatus = document.getElementById('dStatus');
    const dKeeper = document.getElementById('dKeeper');
    const dKeeperRow = document.getElementById('dKeeperRow');
    const dClose = document.getElementById('dClose');

    grid.addEventListener('click', function(e) {
        const point = e.target.closest('.light-point');
        if (!point) return;

        const isTokyo = point.dataset.isTokyo === 'true';
        const isAnxi = point.dataset.isAnxi === 'true';
        const name = point.dataset.name;
        const status = point.dataset.status;

        // 东京 → 跳转 page18
        if (isTokyo) {
            window.location.href = 'page18.html';
            return;
        }

        // 安溪 → 显示特殊信息
        if (isAnxi) {
            dTitle.textContent = '▸ ' + name;
            dStatus.textContent = '💀 已熄';
            dKeeper.textContent = '灯虽熄，愿未灭。后人可续。';
            dKeeperRow.style.display = 'flex';
            overlay.classList.add('visible');
            return;
        }

        // 其他城市
        const statusText = (status === 'extinct') ? '💀 灯灭' : '🔥 灯燃';
        const keeper = getRandomKeeper();
        dTitle.textContent = '▸ ' + name;
        dStatus.textContent = statusText;
        dKeeper.textContent = keeper;
        dKeeperRow.style.display = 'flex';
        overlay.classList.add('visible');
    });

    dClose.addEventListener('click', () => overlay.classList.remove('visible'));
    overlay.addEventListener('click', (e) => { if (e.target === overlay) overlay.classList.remove('visible'); });
    document.addEventListener('keydown', (e) => { if (e.key === 'Escape') overlay.classList.remove('visible'); });

    // =============================================================
    // 4. 控制台
    // =============================================================
    const lit = SITES.filter(s => s.status === 'lit').length;
    const ext = SITES.filter(s => s.status === 'extinct').length;
    console.log(`✦ 天下舆图 ✦ 总城: ${SITES.length} 燃灯: ${lit} 灯灭: ${ext}`);
    console.log('—— 点击东京跳转page18，点击安溪显示特殊信息 ——');
