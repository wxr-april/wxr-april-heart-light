            container.appendChild(el);

            const speed = (Math.random() * 0.9 + 0.3) * (Math.random() > 0.5 ? 1 : -1);
            items.push({
                el: el,
                y: y,
                speed: speed,
                height: 0
            });
        });

        requestAnimationFrame(() => {
            items.forEach(item => {
                item.height = item.el.offsetHeight || 30;
            });
        });

        let lastTime = 0;

        function animate(time) {
            const delta = time - lastTime;
            lastTime = time;
            const step = Math.min(delta, 30);

            const currentWinH = window.innerHeight;
            const currentWinW = window.innerWidth;
            const currentCols = Math.max(10, Math.floor(currentWinW / colWidth));

            items.forEach(item => {
                const h = item.height || 30;
                item.y += item.speed * (step / 16);
                item.el.style.top = item.y + 'px';

                if (item.speed > 0) {
                    if (item.y + h < -10) {
                        item.y = currentWinH + 5 + Math.random() * 15;
                        if (Math.random() > 0.6) {
                            item.speed = (Math.random() * 0.9 + 0.3) * (Math.random() > 0.5 ? 1 : -1);
                        }
                        const newCol = Math.floor(Math.random() * currentCols);
                        item.el.style.left = (newCol * colWidth + (Math.random() * 12 - 6)) + 'px';
                        if (Math.random() > 0.9) {
                            item.el.classList.toggle('highlight');
                        }
                    }
                } else {
                    if (item.y > currentWinH + 10) {
                        item.y = -5 - Math.random() * 15;
                        if (Math.random() > 0.6) {
                            item.speed = (Math.random() * 0.9 + 0.3) * (Math.random() > 0.5 ? 1 : -1);
                        }
                        const newCol = Math.floor(Math.random() * currentCols);
                        item.el.style.left = (newCol * colWidth + (Math.random() * 12 - 6)) + 'px';
                        if (Math.random() > 0.9) {
                            item.el.classList.toggle('ash');
                        }
                    }
                }
            });

            requestAnimationFrame(animate);
        }

        requestAnimationFrame(animate);

        // 3. 上传功能

        const fileInput = document.getElementById('fileInput');
        const statusMsg = document.getElementById('statusMsg');

            if (this.files && this.files.length > 0) {
                const file = this.files[0];
                if (!file.type.startsWith('image/')) {
                    statusMsg.textContent = '请上传图片（天竺语愿纸照片）';
                    statusMsg.className = 'status-msg';
                    return;
                }
                const reader = new FileReader();
                reader.onload = function(ev) {
                    statusMsg.innerHTML = '愿 纸 已 载 入 灯 谱 · 青 烟 升 腾 · 灯 已 燃';
                    statusMsg.className = 'status-msg success';
                    uploadCard.classList.add('accepted');
                    fileInput.disabled = true;
                    document.body.style.transition = 'background 0.6s';
                    document.body.style.backgroundColor = '#0d0500';
                    setTimeout(() => {
                        document.body.style.backgroundColor = '#000000';
                    }, 700);

                    // 1.8秒后跳转 page16b
                    setTimeout(function() {
                        window.location.href = 'page16b.html';
                    }, 1800);
                };
                reader.readAsDataURL(file);
            } else {
                statusMsg.textContent = '未选择愿纸，灯盏虚位。';
                statusMsg.className = 'status-msg';
            }
        });

        document.querySelector('.file-upload-wrapper').addEventListener('click', function(e) {
            if (fileInput.disabled) {
                statusMsg.innerHTML = '🕯 灯 已 燃 · 勿 再 扰';
                statusMsg.className = 'status-msg success';
            }
        });
